      // Charts initialization
      function updateCharts() {
        // Monthly comparison chart
        const monthlyCtx = document.getElementById('monthly-chart').getContext('2d');
        
        // Group expenses by month
        const monthlyData = {};
        const now = new Date();
        const currentMonth = now.getMonth();
        const currentYear = now.getFullYear();
        
        // Initialize last 6 months
        for (let i = 5; i >= 0; i--) {
          const month = new Date(currentYear, currentMonth - i, 1);
          const monthKey = `${month.getMonth() + 1}/${month.getFullYear()}`;
          monthlyData[monthKey] = 0;
        }
        
        // Sum expenses by month
        expenses.forEach(expense => {
          const parts = expense.date.split('/');
          const month = parseInt(parts[1]);
          const year = parseInt(parts[2]);
          const monthKey = `${month}/${year}`;
          
          if (monthlyData[monthKey] !== undefined) {
            monthlyData[monthKey] += expense.amount;
          }
        });
        
        // Convert to arrays for Chart.js
        const months = Object.keys(monthlyData);
        const amounts = Object.values(monthlyData);
        
        // Format months for display
        const displayMonths = months.map(m => {
          const [month, year] = m.split('/');
          const monthNames = ['Ene', 'Feb', 'Mar', 'Abr', 'May', 'Jun', 'Jul', 'Ago', 'Sep', 'Oct', 'Nov', 'Dic'];
          return `${monthNames[parseInt(month) - 1]} ${year}`;
        });
        
        if (window.monthlyChart) {
          window.monthlyChart.destroy();
        }
        
        window.monthlyChart = new Chart(monthlyCtx, {
          type: 'line',
          data: {
            labels: displayMonths,
            datasets: [{
              label: 'Total Gastos Mensuales',
              data: amounts,
              backgroundColor: 'rgba(102, 126, 234, 0.2)',
              borderColor: 'rgba(102, 126, 234, 1)',
              borderWidth: 2,
              pointBackgroundColor: 'rgba(102, 126, 234, 1)',
              tension: 0.4
            }]
          },
          options: {
            scales: {
              y: {
                beginAtZero: true,
                ticks: {
                  callback: function(value) {
                    return '$' + value.toLocaleString('es-CL');
                  }
                }
              }
            },
            plugins: {
              tooltip: {
                callbacks: {
                  label: function(context) {
                    return '$' + context.raw.toLocaleString('es-CL');
                  }
                }
              }
            }
          }
        });
        
        // Category breakdown chart
        const categoryCtx = document.getElementById('category-chart').getContext('2d');
        
        // Group expenses by category
        const categoryData = {};
        
        expenses.forEach(expense => {
          if (!categoryData[expense.category]) {
            categoryData[expense.category] = 0;
          }
          categoryData[expense.category] += expense.amount;
        });
        
        // Convert to arrays for Chart.js
        const categories = Object.keys(categoryData);
        const categoryAmounts = Object.values(categoryData);
        
        // Format categories for display (capitalize)
        const displayCategories = categories.map(c => c.charAt(0).toUpperCase() + c.slice(1));
        
        // Generate colors for each category
        const colors = [
          'rgba(102, 126, 234, 0.8)',
          'rgba(118, 75, 162, 0.8)',
          'rgba(237, 100, 166, 0.8)',
          'rgba(246, 173, 85, 0.8)',
          'rgba(90, 206, 132, 0.8)',
          'rgba(22, 160, 133, 0.8)',
          'rgba(243, 156, 18, 0.8)',
          'rgba(231, 76, 60, 0.8)'
        ];
        
        if (window.categoryChart) {
          window.categoryChart.destroy();
        }
        
        window.categoryChart = new Chart(categoryCtx, {
          type: 'pie',
          data: {
            labels: displayCategories,
            datasets: [{
              data: categoryAmounts,
              backgroundColor: colors.slice(0, categories.length),
              borderWidth: 1
            }]
          },
          options: {
            plugins: {
              legend: {
                position: 'right'
              },
              tooltip: {
                callbacks: {
                  label: function(context) {
                    const label = context.label || '';
                    const value = context.raw;
                    const percentage = ((value / categoryAmounts.reduce((a, b) => a + b, 0)) * 100).toFixed(1);
                    return `${label}: $${value.toLocaleString('es-CL')} (${percentage}%)`;
                  }
                }
              }
            }
          }
        });
      }
      
      // Budget alert check
      function checkBudget() {
        // Calculate total expenses for current month
        const now = new Date();
        const currentMonth = now.getMonth() + 1;
        const currentYear = now.getFullYear();
        
        let monthlyTotal = 0;
        
        expenses.forEach(expense => {
          const parts = expense.date.split('/');
          const expenseMonth = parseInt(parts[1]);
          const expenseYear = parseInt(parts[2]);
          
          if (expenseMonth === currentMonth && expenseYear === currentYear) {
            monthlyTotal += expense.amount;
          }
        });
        
        // Check if over budget
        if (monthlyTotal > budgetLimit) {
          document.getElementById('budget-limit').textContent = budgetLimit.toLocaleString('es-CL', { style: 'currency', currency: 'CLP' });
          document.getElementById('current-spend').textContent = monthlyTotal.toLocaleString('es-CL', { style: 'currency', currency: 'CLP' });
          document.getElementById('budget-alert').classList.remove('hidden');
        } else {
          document.getElementById('budget-alert').classList.add('hidden');
        }
      }
